SQL QUERY EXAMPLE

select concat(e.first_name, ' ', e.last_name) as employee_name,sum(case when a.attendance_status = 'Absent' then 1 else 0 end) as No_of_absences
from employee as e
join attendance as a on e.employee_id = a.employee_id
group by e.employee_id
order by No_of_absences desc,limit 5;
